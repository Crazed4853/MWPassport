function populateTeacherStudentAssistSelect() { const db = loadData(); const select = document.getElementById('teacherStudentAssistSelect'); if (!select || !currentUser) return; // 1. Get student IDs from the teacher's course passports (Course Review Matrix) const courseStudentIds = db.passports .filter(p => p.teacher === currentUser.name) .map(p => p.studentId); // 2. Get student IDs from assigned advisees (Teacher Advisor Matrix) const adviseeStudentIds = db.students .filter(s => s.teacherAdvisorId === currentUser.id) .map(s => s.id); // 3. Combine unique student IDs const allowedStudentIds = new Set([...courseStudentIds, ...adviseeStudentIds]); // 4. Filter and sort matching students const filteredStudents = db.students .filter(s => allowedStudentIds.has(s.id)) .sort((a, b) => a.lastName.localeCompare(b.lastName) || a.firstName.localeCompare(b.firstName)); // 5. Populate dropdown if (filteredStudents.length === 0) { select.innerHTML = ''; return; } select.innerHTML = filteredStudents .map(s => ``) .join(''); }
